STUDENT:

SID   INT         UNSIGNED NOT NULL PRIMARY KEY
SNAME VARCHAR(20)          NOT NULL
PHONE BIGINT      UNSIGNED NOT NULL UNIQUE CHECK(LENGTH(PHONE)=10)


CREATE TABLE STUDENT
(
 SID INT UNSIGNED NOT NULL PRIMARY KEY,
 SNAME VARCHAR(20) NOT NULL,
 PHONE BIGINT UNSIGNED NOT NULL UNIQUE CHECK(LENGTH(PHONE)=10)
);


TO VIEW TABLE STRUCTURE:
------------------------
SYNTAX:

DESC TABLE_NAME;

 
CUSTOMER:

CID     INT                   UNSIGNED NN   PRIMARY KEY AUTO_INCREMENT
CNAME   VARCHAR(30)                    NN
EMAIL   VARCHAR(20)                    NULL UNIQUE
PHONE   BIGINT                UNSIGNED NN   UNIQUE CHECK(LENGTH(PHONE)=10)
GENDER  ENUM("MALE","FEMALE")          NN
ADDRESS TEXT                           NN
BALANCE DECIMAL(5,2)          UNSIGNED NN   DEFAULT "100"
SID     INT                   UNSIGNED NULL FOREIGN KEY

CREATE TABLE CUSTOMER
(
 CID INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
 CNAME VARCHAR(30) NOT NULL,
 EMAIL VARCHAR(20) NULL UNIQUE,
 PHONE BIGINT UNSIGNED NOT NULL UNIQUE CHECK(LENGTH(PHONE)=10),
 GENDER ENUM("MALE","FEMALE") NOT NULL,
 ADDRESS TEXT NOT NULL,
 BALANCE DECIMAL(5,2) UNSIGNED NOT NULL DEFAULT "100",
 SID INT UNSIGNED NULL,
 FOREIGN KEY(SID)REFERENCES STUDENT(SID)
);

PRODUCT:

PID      INT          UNSIGNED NN PRIMARY KEY AUTO_INCREMENT
PNAME    VARCHAR(10)           NN
PRICE    DECIMAL(5,2) UNSIGNED NN
QUANTITY INT          UNSIGNED NN DEFAULT "5"


CREATE TABLE PRODUCT
(
PID INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
PNAME VARCHAR(10) NOT NULL,
PRICE DECIMAL(5,2) UNSIGNED NOT NULL,
QUANTITY INT UNSIGNED NOT NULL DEFAULT "5"
);

ALTER:
------
TO MODIFY TABLE STRUCTURE.

1.TO ADD NEW COLUMN_NAME:
------------------------
SYNTAX

ALTER TABLE TABLE_NAME
ADD NEW_COLUMN_NAME NEW_DATATYPE SIGNED/UNSIGNED NULL/NOT NULL;

STUDENT:

EMAIL VARCHAR(20) NULL


ALTER TABLE STUDENT
ADD EMAIL VARCHAR(20) NULL;

2.TO ADD NEW_COLUMN_NAME AFTER PARTICULAR COLUMN
------------------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD NEW_COLUMN_NAME NEW_DATATYPE SIGNED/UNSIGNED NULL/NOT NULL AFTER EXISTING_COLUMN_NAME;


STUDENT:

BALANCE DECIMAL(4,1) UNSIGNED NOT NULL----->PHONE

ALTER TABLE STUDENT
ADD BALANCE DECIMAL(4,1) UNSIGNED NOT NULL AFTER PHONE;

3.TO ADD NEW_COLUMN_NAME AT FIRST POSITION:
------------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD NEW_COLUMN_NAME NEW_DATATYPE SIGNED/UNSIGNED NULL/NOT NULL FIRST;

STUDENT:
--------
AGE INT UNSIGNED NOT NULL----->FIRST

ALTER TABLE STUDENT
ADD AGE INT UNSIGNED NOT NULL FIRST;

4.TO DROP COLUMN_NAME:
----------------------
SYNTAX:

ALTER TABLE TABLE_NAME
DROP EXISTING_COLUMN_NAME;

STUDENT:

EMAIL

ALTER TABLE STUDENT
DROP EMAIL;

5.TO CHANGE DATATYPE:
---------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME NEW_DATATYPE SIGNED/UNSIGNED NULL/NOT NULL;

ALTER TABLE STUDENT
MODIFY SNAME TEXT NOT NULL;

-------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME EXISTING_DATATYPE SIGNED/UNSIGNED  NULL/NOT NULL;

ALTER TABLE STUDENT

---------

RULES FOR ALIASING:
-------------------

1.WITH OR WITHOUT USING AS KEYWORD WE CAN WRITE ALIAS NAME.

SELECT SAL AS SALARY
FROM EMPS;

SELECT SAL SALARY
FROM EMPS;

SELECT SAL*12  ANNUAL_SALARY
FROM EMPS;

SELECT SAL*12  "ANNUAL SALARY"
FROM EMPS;

---------

-------

CHARACTERISTICS OF ORDER BY CLAUSE:
----------------------------------
1.IT IS LAST EXECUTABLE CLAUSE IN A QUERY.
2.IT EXECUTES AFTER SELECT CLAUSE.
3.BY DEFAULT IT WILL CONSIDER ASCENDING ORDER FOR THE COLUMN.
4.NORMALLY IN ALL THE TABLES RECORDS ARE ARRENGED IN ASCENDING ORDER BASED 
 PRIMARY KEY COLUMN.
5.WE CAN USE ALIAS NAME INSIDE ORDER BY CLAUSE.
6.WE CAN USE MULTIPLE COLUMNS INSIDE ORDER BY CLAUSE BUT IT WILL GIVE THE PRIORITY FOR FIRST COLUMN IF VALUES ARE SAME IN THE FIRST COLUMN THEN IT WILL GIVE THE PRIORITY FOR SECOND COLUMN.

SELECT SAL,LID
FROM EMPS
ORDER BY SAL DESC,LID DESC;

ORDER BY EXECUTION:
--------------------
1.FROM
2.WHERE
3.GROUP BY 
4.HAVING
5.SELECT
6.ORDER BY

DRAWBACK OF SUBQUERY:
---------------------
WE CANT RETRIVE DATA FROM MULTIPLE TABLES SIMULTANEOUSELY.


JOINS:
------
IT IS USED TO RETRIEVE DATA FROM MULTIPLE TABLES SIMULTANEOUSELY.

TYPES OF JOINS:
---------------
1.CROSS JOIN/CARTESIAN JOIN
2.INNER JOIN
3.OUTER JOIN:
             i. LEFT OUTER JOIN/LEFT JOIN
            ii. RIGHT OUTER JOIN/RIGHT JOIN
4.SELF JOIN
5.NATURAL JOIN

1.CROSS JOIN:
-------------
IT IS USED TO DISPLAY MERGED RECORDS FROM MULTIPLE TABLES.

ANSI: AMERICAN NATIONAL STANDARD INSTITUTE

SYNTAX:

SELECT COLUMN_NAME/EXPRESSION
FROM TABLE_NAME T1 CROSS JOIN TABLE_NAME T2;


IF WE USE CROSS JOIN,

THE TOTAL NUMBER OF COLUMNS IN RESULTANT TABLE: C1+C2
THE TOTAL NUMBER OF ROWS IN RESULTANT TABLE   : R1*R2

DRAWBACK OF CROSS JOIN:
-----------------------
IT WILL DISPLAY MORE NUMBER OF UNMATCHED RECORDS COMPARE TO MATCHED RECORDS.

SELECT L.LID,E.LID
FROM LOCATIONS L CROSS JOIN EMPS E;

INNER JOIN:
-----------
IT IS USED TO DISPLAY ONLY MATCHED RECORDS FROM DIFFERENT TABLES.

SYNTAX:
-------

SELECT COLUMN_NAME/EXPRESSION
FROM TABLE_NAME T1 INNER JOIN TABLE_NAME T2
ON JOIN_CONDITION;

ON:IT IS A KEYWORD USED TO PASS JOIN CONDITION.

JOIN CONDITION:IT IS USED TO CONNECT THE TABLES.

SYNTAX:

T1.COLUMN_NAME=T2.COLUMN_NAME

RANKING FUNCTIONS/WINDOW FUNCTIONS
-----------------------------------
IT IS USED TO ASSIGN THE RANKS FOR ALL THE RECORDS IN A TABLE.

TYPES OF RANKING FUNCTION:
--------------------------
1.ROW_NUMBER()
2.RANK()
3.DENSE_RANK()

GENERAL SYNTAX FOR RANKING FUNCTIONS:
-------------------------------------

SELECT RANKING_FUNCTION() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

OVER:IT IS A KEYWORD USED TO PASS RANKING FUNCTIONS

PARTITION BY:IT IS USED TO CREATE GROUPS BUT X

1.ROW_NUMBER():
---------------
IT IS USED TO ASSIGN UNIQUE RANKS FOR THE RECORDS IN A TABLE.

SYNTAX:

SELECT ROW_NUMBER() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,ROW_NUMBER() OVER(PARTITION BY JOB ORDER BY SAL DESC) AS RANKS
FROM EMPS;

SELECT FNAME,JOB,SAL,ROW_NUMBER() OVER(ORDER BY SAL DESC) AS RANKS
FROM EMPS;

DRAWBACK:
---------
IT WILL ASSIGN UNIQUE RANKS FOR TIED ROWS ALSO.

2.RANK():
---------
IT IS USED TO ASSIGN RANKS FOR THE RECORDS IN TABLE.
>IT WILL ASSIGNE SAME RANKS FOR TIED ROWS.

SYNTAX:

SELECT RANK() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,RANK() OVER(ORDER BY SAL DESC) AS RANKS
FROM EMPS;

DRAWBACK:
---------
AFTER ASSIGNING SAME RANKS IT SKIPS NEXT RANKING NUMBER.

3.DENSE_RANK()
--------------
IT IS USED TO ASSIGN RANKS FOR THE RECORDS IN A TABLE.
>IT WILL ASSIGN SAME RANK FOR TIED ROWS AND MAINTAINS NEXT RANKING NUMBERS IN SEQUENTIAL ORDER.

SYNTAX:

SELECT DENSE_RANK() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,DENSE_RANK() OVER(ORDER BY SAL DESC) AS RANKS
FROM EMPS;

1.IS IT POSSIBLE TO CREATE DUPLICATE TABLE?
  YES.

SYNTAX:

CREATE TABLE NEW_TABLE_NAME(SELECT * FROM EXISTING_TABLE_NAME);

EMPS1:
EMPS:

2.is IT POSSIBLE TO CREATE DUPLICATE TABLE WITHOUT RECORDS???

YES...

SYNTAX:
-------
CREATE TABLE NEW_TABLE_NAME(SELECT * FROM EXISTING_TABLE LIMIT 0);

EMPS2:
EMPS(WITHOUT ROWS):

CREATE TABLE EMPS2(SELECT * FROM EMPS LIMIT 0);

3.is IT POSSIBLE TO ADD ROWS FROM ONE TABLE TO ANOTHER TABLE??

YES..IF TABLES STRUCTURE IS SAME

SYNTAX:

VIEW:
----
>IT IS A VIRTUAL TABLE.
>IT DOES'NT OCCUPY ANY MEMORY INSIDE DATABASE.
>IT IS USED TO OVERCOME THE DRAWBACK OF SUBTABLE CONCEPT.

SYNTAX:TO CREATE VIEW:
-----------------------
CREATE VIEW VIEW_NAME AS(DQL QUERY);

EMPS:
WAITER_VIEW:WAITERS DATA FROM EMPS TABLE

CREATE VIEW WAITER_VIEW AS(SELECT * FROM EMPS WHERE JOB="WAITER");


>VIEW CONTAINS DQL QUERIES.
>MODIFICATION PERFORMED IN MAIN TABLE WILL GET AFFECTED AUTOMATICALLY INSIDE VIEW

DIFFERENCE BETWEEN SUBTABLE AND VIEW:
-------------------------------------


INDEX:
------

WHAT IS INDEX?
--------------
IT IS A DATABASE OBJECT WHICH IS USED TO FIND ROWS FASTER WITHOUT SCANNING ENTIRE TABLE.

SYNTAX TO CREATE INDEX:
-----------------------

CREATE INDEX INDEX_NAME ON TABLE_NAME(COLUMN_NAME);

SYNTAX TO VIEW ALL THE INDEXES OF ONE TABLE:
--------------------------------------------
SHOW INDEX FROM TABLE_NAME;

SHOW INDEX FROM EMPS;

BY DEFAULT UNIQUE,FOREIGN KEY AND PRIMARY KEY COLUMNS ARE INDEX COLUMNS.

CREATE INDEX SAL_INDEX ON EMPS(SAL);

SYNTAX TO DROP INDEX:
---------------------

DROP INDEX INDEX_NAME ON TABLE_NAME;

DROP INDEX SAL_INDEX ON EMPS;

SYNTAX TO DROP VIEW:
--------------------

DROP VIEW VIEW_NAME;

DROP VIEW WAITER_VIEW;
